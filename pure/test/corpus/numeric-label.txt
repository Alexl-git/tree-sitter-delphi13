interface
implementation
procedure P(N: Integer);
label 99, 7, Done;
begin
  if N > 1 then
    goto 7;
  if N > 0 then
    goto 99;
  goto Done;
7: N := 3;
Done:
99: N := 4;
end;
end.

implementation

begin
  case x of
    1: x := 2;
    2: x := 3;
  end;
