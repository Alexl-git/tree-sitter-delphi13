end;
{$ENDIF}
{$IFDEF CPUX86}
asm
        MOV     EAX,[EDX].TMethod.Code

{$ENDIF CPUX86}

begin
end;
end.

(root
  (unit
    (kUnit)
    (moduleName (identifier))
    (interface (kInterface))
    (implementation
      (kImplementation)
      (defProc
        header: (declProc (kProcedure) name: (identifier))
        (pp_open)
        body: (block (kBegin) (kEnd)))
      (pp_end_only)
      (pp_open)
      (asm (kAsm) (asmBody) (kEnd))
      (pp_end_only)
      (defProc
        header: (declProc (kProcedure) name: (identifier))
        body: (block (kBegin) (kEnd))))
    (kEnd)
    (kEndDot)))
